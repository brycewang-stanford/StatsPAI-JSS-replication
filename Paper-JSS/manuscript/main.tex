%% Code listing style approximating JSS Sinput / Soutput.  We retain
%% listings (rather than jss.cls's fancyvrb CodeInput) for the numbered,
%% cross-referenceable captions the text relies on, but the rendering is
%% deliberately verbatim-equivalent:
%%
%%   columns=fullflexible + keepspaces  -- listings' default [c]fixed mode
%%     pads every glyph into a fixed box, which spreads identifiers into
%%     "sp . datasets . card_1995 ()" in the PDF and makes the code
%%     uncopyable.  fullflexible uses the typewriter font's natural
%%     widths, so a listing sets exactly as \verb would.
%%   upquote  -- straight ' and " so a reader can paste a listing into a
%%     Python prompt without silently getting curly quotes.
%%   no language=/keywordstyle=  -- JSS does not syntax-highlight code;
%%     bolding Python keywords was a house-style deviation.
%%
%% Every printed listing is executed verbatim by
%% replication/scripts/listings_execute_audit.py, so a listing that stops
%% being runnable fails `make verify` rather than reaching a reviewer.
\definecolor{codebg}{gray}{0.96}
\lstdefinestyle{Sinput}{
  basicstyle=\ttfamily\small,
  backgroundcolor=\color{codebg},
  frame=single, framesep=4pt, framerule=0pt,
  columns=fullflexible,
  keepspaces=true,
  upquote=true,
  breaklines=true,
  breakautoindent=false,
  breakindent=2em,
  postbreak={\mbox{$\hookrightarrow$\space}},
  showstringspaces=false,
  aboveskip=1.2\medskipamount,
  belowskip=0.6\medskipamount,
}
\lstdefinestyle{Soutput}{
  style=Sinput,
  backgroundcolor=\color{codebg!60!white},
}

\title{\statspai{}: Validation-Tiered \proglang{Python} Workflows for
       Causal Inference}
\Plaintitle{StatsPAI: Validation-Tiered Python Workflows for Causal Inference}
\Shorttitle{\statspai{}: Validation-Tiered \proglang{Python} Causal Inference}

\Abstract{
\statspai{} is a \proglang{Python} package for applied causal inference
and econometrics. Every registered function carries an evidence tier that
a user or an agent can query at call time. This article asks what such a
label certifies, and describes a design that makes the answer checkable.
Four links connect a label to the user, and each is verified
mechanically. They are the implementation a parity row exercised, traced
to the source files on its path; the entry point the evidence is
credited to; the configuration and output the row compared, including
whether a standard error was tested as well as the point estimate; and
whether each automated check applies to the fit. Stochastic estimators
are graded by seed-replicated equivalence against an explicit margin.
The evidence covers a twelve-estimator core. It includes a
\RParityModuleCount{}-module harness against canonical \proglang{R}
implementations on identical data bytes, \RParityStrictCount{} of them
at strict parity, with \proglang{Stata} references for
\StataParityModuleCount{}, plus known-truth recovery, calibrated Monte
Carlo coverage, and timings that keep unfavourable comparisons. Applying
the design found defects that numerical agreement alone had missed,
among them parity rows that compared \proglang{R} with itself and an IV
entry point that silently ignored a requested cluster-robust covariance.
Seven worked examples, executed as printed, and a licence-free
replication path close the article. For the agent-facing interface the
evidence is mechanical and contractual, not a claim of behavioural agent
superiority.
}
